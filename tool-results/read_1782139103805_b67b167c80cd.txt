   107→function slugify(s: string): string {
   108→  return (s || "")
   109→    .toLowerCase()
   110→    .normalize("NFD")
   111→    .replace(/[\u0300-\u036f]/g, "") // strip accents
   112→    .replace(/[^a-z0-9]+/g, "-")
   113→    .replace(/^-+|-+$/g, "")
   114→    .slice(0, 80);
   115→}
   116→
   117→function rowToForm(row: ProductRow): ProductForm {
   118→  let specsText = "";
   119→  const s = row.specs;
   120→  if (Array.isArray(s)) {
   121→    specsText = s
   122→      .map((it: any) => {
   123→        const label =
   124→          (it && typeof it === "object" && (it.label?.fr || it.label?.ar)) ||
   125→          (typeof it === "string" ? it : "");
   126→        const value = it && typeof it === "object" ? it.value ?? "" : "";
   127→        return `${label}|${value}`;
   128→      })
   129→      .join("\n");
   130→  } else if (s && typeof s === "object") {
   131→    specsText = Object.entries(s)
   132→      .map(([k, v]) => `${k}|${v}`)
   133→      .join("\n");
   134→  }
   135→  return {
   136→    id: row.id,
   137→    slug: row.slug || "",
   138→    nom_fr: row.nom_fr || "",
   139→    nom_ar: row.nom_ar || "",
   140→    description_fr: row.description_fr || "",
   141→    description_ar: row.description_ar || "",
   142→    specs: specsText,
   143→    images: Array.isArray(row.images) ? row.images.filter(Boolean) : [],
   144→    pdf_url: row.pdf_url || "",
   145→    brochure_pdf: row.brochure_pdf || "",
   146→    video_url: row.video_url || "",
   147→    category_id: row.category_id || "",
   148→    marque: row.marque || "",
   149→    modele: row.modele || "",
   150→    en_vedette: Boolean(row.en_vedette),
   151→    disponible: row.disponible !== false,
   152→    ordre: row.ordre ?? 0,
   153→    cible: Array.isArray(row.cible) ? row.cible.join(", ") : "",
   154→  };
   155→}

   157→function formToPayload(form: ProductForm) {
   158→  // Parse specs textarea → array of { label: { fr, ar }, value }
   159→  const specs = form.specs
   160→    .split("\n")
   161→    .map((line) => line.trim())
   162→    .filter(Boolean)
   163→    .map((line) => {
   164→      const idx = line.indexOf("|");
   165→      let label = "";
   166→      let value = "";
   167→      if (idx === -1) {
   168→        label = line.trim();
   169→      } else {
   170→        label = line.slice(0, idx).trim();
   171→        value = line.slice(idx + 1).trim();
   172→      }
   173→      return { label: { fr: label, ar: label }, value };
   174→    });
   175→  return {
   176→    slug: form.slug,
   177→    nom_fr: form.nom_fr,
   178→    nom_ar: form.nom_ar,
   179→    description_fr: form.description_fr,
   180→    description_ar: form.description_ar,
   181→    specs,
   182→    images: form.images.filter(Boolean),
   183→    pdf_url: form.pdf_url || null,
   184→    brochure_pdf: form.brochure_pdf || null,
   185→    video_url: form.video_url,
   186→    category_id: form.category_id,
   187→    marque: form.marque,
   188→    modele: form.modele,
   189→    en_vedette: form.en_vedette,
   190→    disponible: form.disponible,
   191→    ordre: form.ordre,
   192→    cible: form.cible
   193→      .split(",")
   194→      .map((s) => s.trim())
   195→      .filter(Boolean),
   196→  };
   197→}

   199→// Upload a single file to the admin upload API. Returns the server-stored
   200→// filename on success (the public URL is derived via getProductImageUrl).
   201→// Used by both the multi-image and single-PDF upload flows.
   202→async function uploadFile(
   203→  file: File,
   204→  bucket: "product-images" | "blog-images"
   205→): Promise<
   206→  { ok: true; filename: string; url: string } | { ok: false; error: string }
   207→> {
   208→  const fd = new FormData();
   209→  fd.append("file", file);
   210→  fd.append("bucket", bucket);
   211→  try {
   212→    const res = await fetch("/api/admin/upload", { method: "POST", body: fd });
   213→    const data = await res.json().catch(() => ({}));
   214→    if (!res.ok || !data?.ok) {
   215→      return { ok: false, error: data?.error || `HTTP ${res.status}` };
   216→    }
   217→    return { ok: true, filename: data.filename, url: data.url };
   218→  } catch (e: any) {
   219→    return { ok: false, error: e?.message || "Erreur réseau" };
   220→  }
   221→}
